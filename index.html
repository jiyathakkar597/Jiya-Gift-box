`); const titleI = document.getElementById('f_title'); const artistI = document.getElementById('f_artist'); const spotifyI = document.getElementById('f_spotify'); const ytI = document.getElementById('f_yt'); const appleI = document.getElementById('f_am'); const msgI = document.getElementById('f_msg'); const save = document.getElementById('saveBtn'); const pvTitle = document.getElementById('pv_title'); const pvArtist = document.getElementById('pv_artist'); const pvMsg = document.getElementById('pv_msg'); const player = document.getElementById('songPreviewPlayer'); // Load existing values titleI.value = ed.title || ''; artistI.value = ed.artist || ''; spotifyI.value = ed.spotify || ''; ytI.value = ed.youtube || ''; appleI.value = ed.apple || ''; msgI.value = ed.msg || ''; function spotifyEmbed(url){ try{ const match = url.match( /open\.spotify\.com\/track\/([A-Za-z0-9]+)/ ); if(!match) return null; return ` `; }catch(e){ return null; } } function youtubeEmbed(url){ try{ let videoId = null; const normal = url.match( /youtube\.com\/watch\?v=([^&]+)/ ); const short = url.match( /youtu\.be\/([^?&]+)/ ); const shorts = url.match( /youtube\.com\/shorts\/([^?&]+)/ ); if(normal) videoId = normal[1]; else if(short) videoId = short[1]; else if(shorts) videoId = shorts[1]; if(!videoId) return null; return ` `; }catch(e){ return null; } } function updateSongPreview(){ const title = titleI.value.trim(); const artist = artistI.value.trim(); const spotify = spotifyI.value.trim(); const youtube = ytI.value.trim(); const apple = appleI.value.trim(); const msg = msgI.value.trim(); pvTitle.textContent = title || 'Untitled song'; pvArtist.textContent = artist; pvMsg.textContent = msg; let embed = null; // Spotify first if(spotify){ embed = spotifyEmbed(spotify); } // Then YouTube if(!embed && youtube){ embed = youtubeEmbed(youtube); } if(embed){ player.innerHTML = embed; } else if(apple){ player.innerHTML = ` 🎵 Open in Apple Music `; } else{ player.innerHTML = `
Add a Spotify, YouTube or Apple Music link to play.
`; } save.disabled = !title; } [ titleI, artistI, spotifyI, ytI, appleI, msgI ].forEach(input=>{ input.addEventListener('input', updateSongPreview); }); updateSongPreview(); save.addEventListener('click', ()=>{ commitItem( 'song', { title: titleI.value.trim(), artist: artistI.value.trim(), spotify: spotifyI.value.trim(), youtube: ytI.value.trim(), apple: appleI.value.trim(), msg: msgI.value.trim() }, titleI.value.trim(), existing && existing.id ); }); document .getElementById('cancelBtn') .addEventListener('click', closeModal); }
